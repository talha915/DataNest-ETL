services:

  spark-master:
    image: apache/spark:3.5.7
    container_name: spark-master
    command: >
      /opt/spark/bin/spark-class
      org.apache.spark.deploy.master.Master
    ports:
      - "8080:8080"
      - "7077:7077"
    volumes:
      - ./:/opt/spark-apps  

  spark-worker-1:
    image: apache/spark:3.5.7
    container_name: spark-worker-1
    depends_on:
      - spark-master
    command: >
      /opt/spark/bin/spark-class
      org.apache.spark.deploy.worker.Worker
      spark://spark-master:7077
    environment:
      - SPARK_WORKER_CORES=2
      - SPARK_WORKER_MEMORY=2G
      - SPARK_LOCAL_DIRS=/mnt/w1disk1,/mnt/w1disk2
    volumes:
      - ./:/opt/spark-apps
      - ./disks/worker1-disk1:/mnt/w1disk1
      - ./disks/worker1-disk2:/mnt/w1disk2

  spark-worker-2:
    image: apache/spark:3.5.7
    container_name: spark-worker-2
    depends_on:
      - spark-master
    command: >
      /opt/spark/bin/spark-class
      org.apache.spark.deploy.worker.Worker
      spark://spark-master:7077
    environment:
      - SPARK_WORKER_CORES=2
      - SPARK_WORKER_MEMORY=2G
      - SPARK_LOCAL_DIRS=/mnt/w2disk1,/mnt/w2disk2
    volumes:
      - ./:/opt/spark-apps
      - ./disks/worker2-disk1:/mnt/w2disk1
      - ./disks/worker2-disk2:/mnt/w2disk2

  spark-client:
    image: apache/spark:3.5.7
    container_name: spark-client
    depends_on:
      - spark-master
      - spark-worker-1
      - spark-worker-2
    volumes:
      - ./:/opt/spark-apps
    command: tail -f /dev/null
    ports:
      - "4040:4040"
